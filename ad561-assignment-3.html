Step 2 – Compute Implied Default Probability

Write a function that takes keyword arguments to calculate default probability (PD). Use this formula:

Create a new column called ‘PD’ to show the default probability for each country. Display your entire dataframe again now, showing that this new column has been included.

def compute_pd(yield_i, riskfree_i_star, recovery_rr):
    return (1 - (1 + riskfree_i_star) / (1 + yield_i)) / (1 - recovery_rr)
risk['PD'] = compute_pd(
    yield_i=risk['Yield_i'],
    riskfree_i_star=risk['RiskFree_i_star'],
    recovery_rr=risk['Recovery_RR'])

def system_level_risk(pd_values):
    return 1 - np.prod(1 - pd_values)
portfolio_risk = system_level_risk(risk_shocked['PD'])

import ipywidgets as widgets
from ipywidgets import interact
def bank_stress_test(selected_bank, bond_decline_pct=0.15, deposit_withdrawal_pct=0.20):
    # pull the selected bank's row
    row = bank[bank['Bank'] == selected_bank].iloc[0]

    assets = row['Assets']
    liabilities = row['Liabilities']
    bond_portfolio = row['Bond_Portfolio']
    deposits = row['Uninsured_Deposits']

    bond_loss = bond_portfolio * bond_decline_pct
    deposit_outflow = deposits * deposit_withdrawal_pct

    updated_assets = assets - bond_loss - deposit_outflow
    updated_equity = updated_assets - liabilities

    solvency = "Solvent" if updated_equity > 0 else "Insolvent"

    print(f"Bank: {selected_bank}")
    print(f"Bond Portfolio Decline: {bond_decline_pct:.0%}")
    print(f"Deposit Withdrawal: {deposit_withdrawal_pct:.0%}")
    print(f"Updated Assets: {updated_assets:.2f}")
    print(f"Updated Equity: {updated_equity:.2f}")
    print(f"Solvency Status: {solvency}")
    plt.figure(figsize=(6,4))
    plt.bar(['Updated Assets', 'Updated Equity'], [updated_assets, updated_equity])
    plt.title(f'Bank Stress Test: {selected_bank}')
    plt.ylabel('Value')
    plt.show()

interact(
